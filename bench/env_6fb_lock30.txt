# Longshot on the ASUS ROG Ally (RC71L, Z1 Extreme): the Hookshot fork's tested setup, without its fork-only switches.
# Ally Z1E: the native mesh path hangs the GPU or reboots (upstream #997). Use the compute mesh fallback (also the options file's mesh_shaders=2).
RX_VK_FORCE_MESH_FALLBACK=1
# main predicts ~65,000 extra pipelines (hours on the Ally): build them for 60 s at most, the rest when first drawn
RX_VK_PREDICT_S=60
# 32 MB image blocks instead of 128: each fresh block costs ~0.5 ms/MB to allocate on the Ally (~27 ms vs ~65 ms per hitch)
RX_VK_IMAGE_BLOCK_MB=32
# Don't load (or save) the 1-2.5 GB pipeline cache file; the AMD driver's own shader cache covers it: +3.5 game fps, better 1% lows,
# ~1/3 less stutter time (interleaved runs). After an AMD driver update or a big Windows update the driver cache can be empty, and
# the first start then rebuilds every shader (~30 min, with a progress screen). To undo: delete this line.
RX_VK_PCACHE_LOAD=0
# --- Optional diagnostics (this fork's patches): they only write a line when something is slow or memory is allocated.
RX_VK_SLOW_LOG=80
# Programs and their pipelines loaded when the game makes them, not all before the title: the 4-5 GB start mode keeps ready (load_default.h), on a box with 7.9 GB of RAM
RX_VK_LOAD=create
# (Ally test) RX_VK_READBACK_LATE=newest
RX_VK_READBACK_LATE=newest,3980,9b040,1b580,8b040
# (Ally test 2026-11-15) + 8b040: the two 0x8b040 readback ranges waited ~16.5 ms each in Town and Kokiri (4090 laptop, RX_VK_LOG_READBACK_RANGES); FIRST_PART=rb gained exactly those
# (Ally test, patch 0004) + capture write-back waits (vk_timeline_wait) end at the first part: with the 8b040 readbacks read late, is the capture wait the next limit?
RX_VK_FENCE_FIRST_PART=cap
# (q22) the play env with HOS_FPS_LOCK=30 (40: patch 0006 paces every 25 ms; 30: patch 0001, 2 vsyncs of 60)
HOS_FPS_LOCK=30
