# 4090 laptop (DESKTOP-K4ELB80, RTX 4090 Laptop + Intel UHD): the Ally's Longshot setup on the Ally's GPU path, for issue 997
# flash work (q4090-chain.ps1). Same switches as the Ally's env_flash_*.txt, plus RX_VK_MIRROR=0: no VRAM mirror of the buffer
# pools, so the GPU reads and writes guest memory in place, as on the Ally's iGPU. Speed is measured on the Ally only.
RX_VK_MIRROR=0
# Ally Z1E: the native mesh path hangs the GPU or reboots (upstream #997). Use the compute mesh fallback (also the options file's mesh_shaders=2).

# main predicts ~65,000 extra pipelines (hours on the Ally): build them for 60 s at most, the rest when first drawn
RX_VK_PREDICT_S=60
# 32 MB image blocks instead of 128 (the Ally's value; kept for the same allocation pattern)
RX_VK_IMAGE_BLOCK_MB=32
# Don't load (or save) the pipeline cache file; the driver's own shader cache covers it (the Ally's value)
RX_VK_PCACHE_LOAD=0
# --- Optional diagnostics: they only write a line when something is slow or memory is allocated.
RX_VK_SLOW_LOG=80
# Programs and their pipelines loaded when the game makes them, not all before the title (the Ally's value)
RX_VK_LOAD=create
# rball: the Ally's current setting (+7 %, 0 flashes)
RX_VK_READBACK_LATE=newest,3980,9b040,1b580
# (diagnostic) 400 FenceSync / SyncWait lines with their caller (lr) from present 6000 on (~100 s at 60 fps: past the load),
# plus nvn_native's 'GPU waits in 5 s' totals; and each readback wait's recordings (RX_VK_LOG_READBACK_WAITS)
HOS_NVN_SYNC_LOG=6000
RX_VK_LOG_READBACK_WAITS=1
