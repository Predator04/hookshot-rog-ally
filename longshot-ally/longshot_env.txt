# Longshot on the ROG Ally (Ryzen Z1 Extreme, 16 GB, UMA frame buffer 4 GB): the tested settings.
# The game reads this file from its folder at every start (restart after a change). A line starting with # does nothing.
# Tested on Longshot v3.0.0-rc.7 + this fork's frame cap patch (0007). Details: README.md beside this file.

# The Ally's GPU hangs or the PC restarts with the native mesh shaders: run them as compute shaders. Keep this line: it's
# the only thing that chooses this (Longshot dropped the menu row, and its automatic rule doesn't know the Ally's GPU).
# AMD's current driver doesn't offer mesh shaders on the Ally anyway; the line keeps you safe if a driver update does.
RX_VK_FORCE_MESH_FALLBACK=1

# No pipeline cache file: the AMD driver's own shader cache covers it (on an earlier Longshot this gave more fps and fewer
# stutters on the Ally). The driver keeps its cache by the game's folder: keep the folder where it is.
RX_VK_PCACHE_LOAD=0

# CPU reads of GPU results from the newest submission don't wait for it: they read the values a frame late (+13 % on an
# earlier Longshot on the Ally). With this Longshot's early reads (on by default here) this alone is as fast as the older
# list of sizes. If you see a white flash or an object popping for a frame, delete this line.
RX_VK_READBACK_LATE=newest

# Frame generation by the renderer's own maker. Longshot's default here, AMD FSR 3.1 frame generation, looks better but
# costs about 11 % of the game's frame rate on the Ally (3.9 ms more GPU time a frame). Delete this line to try FSR 3.1.
RX_VK_FSR_FG=0

# A steady 36 fps (needs this fork's build: official Longshot takes only 30 or 61 and up here).
# 36 holds everywhere tested; 40 dips in Hyrule Field's heaviest views. Delete the line for no cap.
HOS_FPS_MAX=36
